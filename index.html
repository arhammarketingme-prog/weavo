let isAuthProcessing = false; // गुगल लॉगिन बटण क्लिक async function signInWithGoogle() { const errEl = $('error'); if (errEl) errEl.textContent = ''; const gBtn = $('google-login-btn'); if (gBtn) { gBtn.disabled = true; gBtn.innerHTML = `पडताळणी सुरू आहे, कृपया थांबा...`; } const cleanUrl = window.location.origin + window.location.pathname; const { error } = await sb.auth.signInWithOAuth({ provider: 'google', options: { redirectTo: cleanUrl } }); if (error) { if (errEl) errEl.textContent = error.message; if (gBtn) { gBtn.disabled = false; gBtn.innerHTML = `Continue with Google`; } } } if ($('google-login-btn'))$('google-login-btn').onclick = signInWithGoogle; // ऑप्टिमाइझ केलेले फास्ट सेशन हँडलर async function setupUserSession(user) { if (!user || isAuthProcessing || me) return; isAuthProcessing = true; // १. त्वरित UI सुरू करा (युझरला थांबावे लागणार नाही) let initialUsername = user.user_metadata?.full_name?.replace(/\s+/g, '').toLowerCase() || user.email?.split('@')[0] || 'user'; me = { id: user.id, username: initialUsername }; // २. URL मधील हॅश स्वच्छ करा if (window.location.hash && window.location.hash.includes('access_token')) { window.history.replaceState({}, document.title, window.location.pathname + window.location.search); } // ३. त्वरित मुख्य ॲप उघडा enterApp(); // ४. बॅकग्राउंडमध्ये प्रोफाईल सिंक करा (UI अडकणार नाही) try { const { data: profile } = await sb.from('profiles').select('username').eq('id', user.id).maybeSingle(); if (profile?.username) { me.username = profile.username; if ($('me-name'))$('me-name').textContent = '@' + me.username; } else { await sb.from('profiles').upsert({ id: user.id, username: initialUsername }); } } catch (e) { console.warn("Profile sync in background:", e); } finally { isAuthProcessing = false; } } // ऑटो-सेशन श्रोता (फक्त एकदाच ट्रिगर होईल) sb.auth.onAuthStateChange(async (event, session) => { if (session?.user && !me) { await setupUserSession(session.user); } });
